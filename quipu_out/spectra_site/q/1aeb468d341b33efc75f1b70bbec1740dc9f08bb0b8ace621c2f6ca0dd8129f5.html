% Margin banner IV — El Descenso (the hollow earth)
% Concentric subterranean shells centred on a core: the crust at the rim, the
% nested spheres of Dante's descent, a polar axis falling to the Hyperborean
% inner sun at the world's hollow centre. xelatex 04_subterranean.tex
\documentclass[tikz,border=3mm]{standalone}
\usepackage{fontspec}\setmainfont{Hoefler Text}
\usepackage{xcolor}\usepackage{tikz}\usetikzlibrary{shapes.geometric}
\definecolor{kr}{HTML}{c83727}\definecolor{ky}{HTML}{e8b73a}\definecolor{kb}{HTML}{2b62a8}
\definecolor{kg}{HTML}{2a7a4d}\definecolor{kp}{HTML}{6e3a8a}\definecolor{kgold}{HTML}{c2a76b}
\definecolor{kw}{HTML}{f4ead8}\definecolor{ki}{HTML}{1a1a1a}
\definecolor{kox}{HTML}{6e1a1a}\definecolor{kcrust}{HTML}{8a6a52}\definecolor{kmantle}{HTML}{a9886c}
\begin{document}\begin{tikzpicture}[x=1cm,y=1cm]
  \fill[kw] (-3.6,-15.2) rectangle (3.6,15.2);

  % --- concentric shells, centred low on the core (clipped to the frame) --
  \begin{scope}
    \clip (-3.25,-14.85) rectangle (3.25,14.85);
    \foreach \r/\c/\o in {16/kcrust/0.30, 13.5/kmantle/0.30, 11/kox/0.26, 8.8/kcrust/0.34,
                          6.8/kmantle/0.36, 5.0/kox/0.34, 3.4/kgold/0.40, 2.0/kox/0.5} {
      \fill[\c,opacity=\o] (0,-11.5) circle (\r);
    }
  \end{scope}

  % double gold frame on top
  \draw[kgold,line width=0.6mm] (-3.4,-15.0) rectangle (3.4,15.0);
  \draw[kgold,line width=0.15mm] (-3.25,-14.85) rectangle (3.25,14.85);

  \node[ki,font=\itshape,anchor=north] at (0,14.4) {The Descent};
  \node[ki,font=\scriptsize\itshape,anchor=north] at (0,13.6) {the hollow earth};

  % polar axis falling from the crust to the core
  \draw[kgold,line width=0.7pt,dash pattern=on 2pt off 2pt] (0,12.6) -- (0,-10.0);

  % shell labels along the descent (right side)
  \node[ki,font=\fontsize{6}{7}\selectfont\itshape,anchor=west] at (1.9,11.2) {crust};
  \node[ki,font=\fontsize{6}{7}\selectfont\itshape,anchor=west] at (2.0,5.4) {the circles};
  \node[ki,font=\fontsize{6}{7}\selectfont\itshape,anchor=east] at (-1.9,-1.2) {concentric};
  \node[ki,font=\fontsize{6}{7}\selectfont\itshape,anchor=east] at (-1.9,-1.9) {spheres};

  % Dante — a small robed silhouette descending the axis, mid-fall
  \fill[ki] (0,3.2) .. controls (-0.45,2.4) and (-0.35,1.0) .. (-0.30,0.4)
       -- (0.30,0.4) .. controls (0.35,1.0) and (0.45,2.4) .. (0,3.2) -- cycle;
  \fill[ki] (0,3.45) circle (0.22);                 % head/hood
  \node[ki,font=\fontsize{6}{7}\selectfont\itshape,anchor=west] at (0.5,2.0) {Dante};

  % --- the Hyperborean inner sun at the hollow core ----------------------
  \foreach \a in {0,30,...,330} \draw[ky,line width=0.8pt] (0,-11.5)+(\a:1.0) -- +(\a:1.9);
  \fill[ky] (0,-11.5) circle (0.95);
  \fill[kgold] (0,-11.5) circle (0.6);

        line width=0.2pt, minimum size=0.72cm, inner sep=0] at (0,-11.5) {};
  \node[ki,font=\fontsize{6.5}{7}\selectfont\itshape,anchor=north] at (0,-12.9) {Hyperborea};
  \node[ki,font=\fontsize{6}{7}\selectfont\itshape,anchor=north] at (0,-13.5) {the inner sun};
